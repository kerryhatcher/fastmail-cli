#[tokio::test]
async fn error_401_returns_invalid_token() {

    assert!(matches!(err, Error::InvalidToken(_)), "got {err:?}");
}

#[tokio::test]
async fn error_429_returns_rate_limited() {

mod common;

use wiremock::{Mock, ResponseTemplate, Request};
use wiremock::matchers::{method, path, body_partial_json};
use serde_json::json;

#[tokio::test]
async fn send_email_posts_correct_wire_format() {
    let server = common::start_mock_server().await;

    // 1. Session endpoint
    Mock::given(method("GET")).and(path("/jmap/session"))
        .respond_with(ResponseTemplate::new(200)
            .set_body_raw(common::jmap_session_response(&server.uri()), "application/json"))
        .mount(&server).await;

    // 2. Respond to all /jmap/api/ POSTs by inspecting the methodCalls and routing.
    //    Simplest approach: use a single mock that matches any body and returns the merged fixture.
    //    Real Fastmail returns one response per method call in methodCalls — we can do the same.

    // For the test: respond to the first POST with Mailbox/get + Identity/get + Email/set + EmailSubmission/set
    // response. (send_email in src may issue multiple sequential POSTs; mount separate mocks per call
    // if needed — inspect actual request sequence.)

    // Request assertion: capture body via wiremock's received request list after the call.
    Mock::given(method("POST")).and(path("/jmap/api/"))
        .respond_with(ResponseTemplate::new(200).set_body_raw(
            // merged multi-response fixture — if send_email makes multiple POSTs, mount multiple mocks
            // with .up_to_n_times() or use body_partial_json matchers to route.
            common::load_fixture("jmap_email_set_response.json"),
            "application/json",
        ))
        .mount(&server).await;

    let mut client = common::test_jmap_client(&server);
    client.authenticate().await.expect("auth");

    // Call send_email with actual signature from src/jmap/mod.rs. Verify in that file first.
    // Example (pseudo - adjust to actual API):
    let to = vec![fastmail_cli::models::EmailAddress {
        email: "dest@example.com".to_string(),
        name: Some("Dest".to_string()),
    }];
    let result = client.send_email(
        &to, &[], &[],  // to, cc, bcc
        "Test Subject",
        "Test body content",
        None,  // from
        false, // draft
    ).await;

    assert!(result.is_ok(), "send_email failed: {:?}", result.err());

    // Assert wire format: fetch received requests and inspect.
    let received = server.received_requests().await.expect("received");
    let post_bodies: Vec<serde_json::Value> = received.iter()
        .filter(|r| r.method == wiremock::http::Method::POST)
        .map(|r| serde_json::from_slice(&r.body).expect("valid json body"))
        .collect();

    assert!(!post_bodies.is_empty(), "expected at least one POST");
    // Find an Email/set call in any of the request bodies' methodCalls
    let found_email_set = post_bodies.iter().any(|body| {
        body.get("methodCalls")
            .and_then(|mc| mc.as_array())
            .map(|arr| arr.iter().any(|call| {
                call.get(0).and_then(|v| v.as_str()) == Some("Email/set")
            }))
            .unwrap_or(false)
    });
    assert!(found_email_set, "expected Email/set in request methodCalls; bodies: {:#?}", post_bodies);

    // Verify subject made it into wire format
    let body_str = serde_json::to_string(&post_bodies).unwrap();
    assert!(body_str.contains("Test Subject"), "subject not in wire body");
    assert!(body_str.contains("dest@example.com"), "recipient not in wire body");
}
