mod common;

use std::sync::Arc;
use tokio::sync::Mutex;
use wiremock::{Mock, ResponseTemplate};
use wiremock::matchers::{method, path};
use async_graphql::{Schema, EmptySubscription};

use fastmail_cli::mcp::graphql::AppContext;
// Import the actual schema root types. Verify paths:
// If query::QueryRoot and mutation::MutationRoot are pub, import them.
// Otherwise use a factory fn like fastmail_cli::mcp::graphql::build_schema(ctx).
use fastmail_cli::mcp::graphql::{QueryRoot, MutationRoot};  // ADJUST if not re-exported

#[tokio::test]
async fn graphql_query_executes_with_test_app_context() {
    let server = common::start_mock_server().await;

    // Mock JMAP session + one Mailbox/get call, so a JMAP-backed query succeeds.
    Mock::given(method("GET")).and(path("/jmap/session"))
        .respond_with(ResponseTemplate::new(200)
            .set_body_raw(common::jmap_session_response(&server.uri()), "application/json"))
        .mount(&server).await;

    // Any JMAP API POST returns a plausible Mailbox/get response.
    // (Adjust fixture per the actual query invoked.)
    let mailbox_get_response = r#"{
      "methodResponses":[["Mailbox/get",{"accountId":"u1234abcd","state":"1","list":[{"id":"Mbox-inbox","name":"Inbox","role":"inbox","parentId":null,"sortOrder":0,"totalEmails":0,"unreadEmails":0,"totalThreads":0,"unreadThreads":0,"myRights":{"mayReadItems":true,"mayAddItems":true,"mayRemoveItems":true,"maySetSeen":true,"maySetKeywords":true,"mayCreateChild":true,"mayRename":true,"mayDelete":false,"maySubmit":true},"isSubscribed":true}],"notFound":[]},"c1"]],
      "sessionState":"abc123"
    }"#;
    Mock::given(method("POST")).and(path("/jmap/api/"))
        .respond_with(ResponseTemplate::new(200)
            .set_body_raw(mailbox_get_response, "application/json"))
        .mount(&server).await;

    // Build JMAP client pointed at mock, authenticate up front.
    let mut jmap_client = common::test_jmap_client(&server);
    jmap_client.authenticate().await.expect("authenticate");
    let jmap = Arc::new(Mutex::new(jmap_client));

    // Test AppContext with deterministic HMAC key.
    let ctx = AppContext::new_with_key(Some(jmap), [0u8; 32]);

    // Build schema. If build_schema factory is exposed, use it.
    // Otherwise use async_graphql Schema::build directly:
    let schema = Schema::build(QueryRoot::default(), MutationRoot::default(), EmptySubscription)
        .limit_depth(5)
        .limit_complexity(200)
        .data(ctx)
        .finish();

    // Pick a simple Query. Check src/mcp/graphql/query.rs for an available field.
    // Example: `{ listMailboxes { id name role } }` if that field exists, or a ping/schema introspection query.
    let query = r#"{ listMailboxes { id name role } }"#;
    let resp = schema.execute(query).await;

    assert!(resp.errors.is_empty(), "graphql errors: {:?}", resp.errors);
    let json = serde_json::to_value(&resp.data).expect("serialize data");
    // Assert a mailbox came through
    assert!(json.to_string().contains("Inbox"), "expected Inbox in response: {json}");

    // Also test confirmation_token determinism (no HTTP)
    let token1 = AppContext::new_with_key(None, [0u8; 32]).confirmation_token(&["deleteContact", "abc"]);
    let token2 = AppContext::new_with_key(None, [0u8; 32]).confirmation_token(&["deleteContact", "abc"]);
    assert_eq!(token1, token2, "same key + same inputs must produce same token");
    let token3 = AppContext::new_with_key(None, [1u8; 32]).confirmation_token(&["deleteContact", "abc"]);
    assert_ne!(token1, token3, "different keys must produce different tokens");
}

#[tokio::test]
async fn graphql_introspection_works() {
    // Minimal smoke test: schema exposes its own shape
    let ctx = AppContext::new_with_key(None, [0u8; 32]);
    let schema = Schema::build(QueryRoot::default(), MutationRoot::default(), EmptySubscription)
        .limit_depth(5)
        .limit_complexity(200)
        .data(ctx)
        .finish();
    let resp = schema.execute("{ __schema { queryType { name } } }").await;
    assert!(resp.errors.is_empty(), "introspection errors: {:?}", resp.errors);
    let json = serde_json::to_value(&resp.data).unwrap();
    assert!(json.to_string().contains("queryType"), "expected queryType in introspection");
}
