**tests/carddav_crud.rs**:
```rust
mod common;

use wiremock::{Mock, ResponseTemplate};
use wiremock::matchers::{method, path, path_regex};
use fastmail_cli::carddav::{Contact, ContactEmail};

fn sample_contact() -> Contact {
    Contact {
        id: "contact-1".to_string(),
        name: "Alice Example".to_string(),
        emails: vec![ContactEmail {
            email: "alice@example.com".to_string(),
            label: Some("WORK".to_string()),
        }],
        phones: vec![],
        organization: None,
        title: None,
        notes: None,
        address: None,
        href: None,
        etag: None,
    }
}

#[tokio::test]
async fn carddav_crud_roundtrip() {
    let server = common::start_mock_server().await;
    let addressbook_href = "/dav/addressbooks/user/test@example.com/Default/";

    // 1. Create: PUT with If-None-Match: * -> 201 Created + ETag + Location
    Mock::given(method("PUT"))
        .and(path_regex(r"^/dav/addressbooks/user/test@example.com/Default/[^/]+\.vcf$"))
        .respond_with(ResponseTemplate::new(201)
            .insert_header("ETag", "\"etag-1\"")
            .insert_header("Location", "/dav/addressbooks/user/test@example.com/Default/contact-1.vcf"))
        .expect(2)  // 1 create + 1 update
        .mount(&server).await;

    // 2. Delete: DELETE -> 204
    Mock::given(method("DELETE"))
        .and(path("/dav/addressbooks/user/test@example.com/Default/contact-1.vcf"))
        .respond_with(ResponseTemplate::new(204))
        .expect(1)
        .mount(&server).await;

    let client = common::test_carddav_client(&server);
    let contact = sample_contact();

    // CREATE
    let created = client.create_contact(addressbook_href, &contact).await
        .expect("create_contact");
    assert!(created.href.contains("contact-1.vcf"), "href: {}", created.href);
    assert_eq!(created.etag.as_deref(), Some("\"etag-1\""));

    // UPDATE — use created href + etag on contact
    let mut updated_contact = contact.clone();
    updated_contact.href = Some(created.href.clone());
    updated_contact.etag = created.etag.clone();
    updated_contact.name = "Alice Updated".to_string();
    // Method signature may differ; adjust. If update_contact returns ETag, capture.
    client.update_contact(&updated_contact).await
        .expect("update_contact");

    // DELETE
    client.delete_contact(&created.href, created.etag.as_deref().unwrap_or("")).await
        .expect("delete_contact");

    // Verify request sequencing: at least 2 PUTs + 1 DELETE observed
    let received = server.received_requests().await.expect("received");
    let puts = received.iter().filter(|r| r.method.as_str() == "PUT").count();
    let deletes = received.iter().filter(|r| r.method.as_str() == "DELETE").count();
    assert_eq!(puts, 2, "expected 2 PUTs (create+update), got {puts}");
    assert_eq!(deletes, 1, "expected 1 DELETE, got {deletes}");

    // Assert first PUT had If-None-Match: * header (new contact create)
    let first_put = received.iter().find(|r| r.method.as_str() == "PUT").unwrap();
    let inm = first_put.headers.get("if-none-match")
        .map(|v| v.to_str().unwrap_or(""))
        .unwrap_or("");
    assert!(inm.contains('*'), "first PUT should carry If-None-Match: *, got {inm:?}");
}
```
